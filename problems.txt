       const originalFetch = window.fetch;

   ```python
   import hashlib
   import secrets
   from datetime import timedelta
   from django.utils import timezone
   from django.core.cache import cache
   
   class OTPService:
       OTP_LENGTH = 6
       OTP_TIMEOUT = 300  # 5 minutes
       MAX_ATTEMPTS = 3
       CACHE_PREFIX = 'otp_'
       
       @staticmethod
       def send_otp(phone_number):
           """OTP yuborish"""
           otp = ''.join([str(secrets.randbelow(10)) for _ in range(OTPService.OTP_LENGTH)])
           
           cache_key = f"{OTPService.CACHE_PREFIX}{phone_number}"
           cache_data = {
               'otp': otp,
               'attempts': 0,
               'created_at': timezone.now().isoformat()
           }
           
           cache.set(cache_key, cache_data, OTPService.OTP_TIMEOUT)
           
           # SMS yuborish logic
           return True
       
       @staticmethod
       def verify_otp(phone_number, otp):
           """OTP tekshirish"""
           cache_key = f"{OTPService.CACHE_PREFIX}{phone_number}"
           cache_data = cache.get(cache_key)
           
           if not cache_data:
               return False, "OTP expired"
           
           if cache_data['attempts'] >= OTPService.MAX_ATTEMPTS:
               cache.delete(cache_key)
               return False, "Max attempts exceeded"
           
           if cache_data['otp'] != otp:
               cache_data['attempts'] += 1
               cache.set(cache_key, cache_data, OTPService.OTP_TIMEOUT)
               return False, "Invalid OTP"
           
           cache.delete(cache_key)
           return True, "OTP verified"
   ```
